Scaled Dot-Product Attention
Implement from scratch in NumPy. No PyTorch. Full forward pass including masking.

import numpy as np def attention(Q, K, V, mask=None): d_k = Q.shape[-1] scores = Q @ K.T / np.sqrt(d_k) if mask is not None: scores += mask * -1e9 w = np.exp(scores - scores.max(-1,keepdims=True)) w /= w.sum(-1, keepdims=True) return w @ V

GPT-2 Causal Attention + KV Cache

Click to reveal answer
def causal_attn(x, Wq, Wk, Wv, cache_k=None, cache_v=None): Q,K,V = x@Wq, x@Wk, x@Wv if cache_k is not None: K = np.vstack([cache_k, K]) V = np.vstack([cache_v, V]) scores = Q @ K.T / sqrt(d_k) mask = np.triu(np.ones_like(scores), k=1)*-1e9 w = softmax(scores + mask) return w @ V, K, V
